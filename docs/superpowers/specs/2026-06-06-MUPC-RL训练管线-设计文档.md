
### 2.3 电压仿真（VoltageSimulator 降级模式）

```python
class VoltageSimulator:
    """三相电压简化线路模型（Grid2Op 不可用时的降级方案）。

    物理基础: Q-V 耦合。感性无功吸收 → 电压降低；
             容性无功释放 → 电压升高。
    """

    K_P = 0.05   # 有功灵敏度 (p.u. / S_BASE)
    K_Q = 0.03   # 无功灵敏度 (p.u. / S_BASE)
    S_BASE = 200.0  # kVA（匹配变压器容量，v2.6）
    V_MIN, V_MAX = 0.85, 1.15
    NOISE_STD = 0.005  # 测量噪声标准差
    PHASE_IMBALANCE = 0.003  # 三相不平衡度

    def step(self, p_net: float, q_batt: float,
             prev_va: float, prev_vb: float, prev_vc: float
             ) -> tuple[float, float, float]:
        """
        Args:
            p_net: 净有功功率 = P_pv - P_load_eff + P_batt (kW)
            q_batt: 无功功率 (kVar)
            prev_v*: 上一周期电压
        Returns:
            (va, vb, vc) 三相电压标幺值
        """
        # 公用电压变化量 (基于净有功和无功)
        dv = (self.K_P * p_net + self.K_Q * q_batt) / self.S_BASE

        # 三相分别叠加噪声和不平衡
        va = prev_va + dv + np.random.normal(0, self.NOISE_STD)
        vb = prev_vb + dv + np.random.normal(0, self.NOISE_STD) + self.PHASE_IMBALANCE
        vc = prev_vc + dv + np.random.normal(0, self.NOISE_STD) - self.PHASE_IMBALANCE

        return (
            np.clip(va, self.V_MIN, self.V_MAX),
            np.clip(vb, self.V_MIN, self.V_MAX),
            np.clip(vc, self.V_MIN, self.V_MAX),
        )
```

### 3.2 to_input_vector() — 58 维序列化

```python
def _build_observation(self) -> np.ndarray:
    """构建 58维观测 (多模式 59 维)。对齐下游 MUPC AI 引擎设计文档 v2.5."""
    obs = np.zeros(58 if self.mode != "all" else 59, dtype=np.float32)

    # [0..9] D1: 10 个标量 (含 q_realtime_margin)
    obs[0] = self.soc
    obs[1] = self.data['pv_power'][self.step_idx] * 200 / 1000  # 缩放到 200kW
    obs[2] = self.data['load_power'][self.step_idx]
    obs[3] = self.grid_power
    obs[4] = self.load_rate
    obs[5] = self.battery_power_prev
    obs[6] = self.va
    obs[7] = self.vb
    obs[8] = self.vc
    obs[9] = self._q_realtime_margin  # v2.5 新增

    # [10..24] D2 pv_forecast: 15 维
    pv_fc = self._get_pv_forecast()  # from LSTM or Oracle
    obs[10:25] = pv_fc[:15]

    # [25..39] D2 load_forecast: 15 维
    load_fc = self._get_load_forecast()
    obs[25:40] = load_fc[:15]

    # [41..43] D3: 3 个电价字段
    obs[41] = self.data['current_price'][self.step_idx]
    obs[42] = self.data['next_price'][self.step_idx]
    obs[43] = self.data['tariff_id'][self.step_idx]

    # [44..46] D4: 3 个需量字段
    obs[44] = self.current_demand
    obs[45] = self.contract_demand
    obs[46] = self.peak_demand

    # [47..48] D5: 2 个气象字段
    obs[47] = self.data['ghi'][self.step_idx]
    obs[48] = self.data['temperature'][self.step_idx]

    # [49] D6: dispatch_p_set (None 填 0.0)
    obs[49] = self.data['dispatch_p'][self.step_idx] or 0.0

    # [50] D7: q_realtime_margin
    obs[50] = self._q_realtime_margin

    # [51..56] D7: season_encoding (6维)
    obs[51:57] = self._season_encoding

    # [57] D7: time_period_encoding (1维)
    obs[57] = self._time_period_encoding[0]

    # [58] (可选) mode_id
    if self.mode == "all":
        obs[58] = self._mode_to_id(self.current_mode)

    # 归一化
    obs = self._normalize(obs)
    return obs
```

### 3.3 物理仿真方程

```python
@dataclass
class PhysicsResult:
    p_batt: float       # 电池有功 (kW)
    q_batt: float       # 无功 (kVar)
    load_shedding: float # 切负荷 (kW)
    pv_limit: float      # 光伏限功率比例
    soc_new: float
    grid_power: float
    load_rate: float
    va: float; vb: float; vc: float
    soc_clipped: bool
    constraint_violated: bool

def _physics_step(self, action_clamped: np.ndarray) -> PhysicsResult:
    """一步物理仿真。"""

    # 1. 动作反归一化（3维: P_batt + Load_shedding + Pv_limit）
    p_batt = action_clamped[0] * self.P_BATT_MAX_KW       # [-50, 50] kW（v2.6）
    q_batt = self._q_batt_realtime  # Q 由实时电压调节器闭环（不经过 RL）
    load_shedding = action_clamped[1] * self.LOAD_SHED_MAX_KW  # [0, 60] kW（v2.6）
    pv_limit = action_clamped[2]                           # [0, 1]（v2.6 新增）

    # 2. 有效负荷和光伏
    p_load_raw = self.data['load_power'][self.step_idx]
    p_load_eff = max(0, p_load_raw - load_shedding)   # 切负荷后
    p_pv_raw = self.data['pv_power'][self.step_idx]
    p_pv_eff = p_pv_raw * pv_limit                     # 限功率后

    # 3. SOC 更新 (SAFETY: hard clamp)
    soc_raw = self.soc + (-p_batt * self.DT_HOURS) / self.BATTERY_CAPACITY_KWH
    soc_new = np.clip(soc_raw, self.SOC_MIN, self.SOC_MAX)
    soc_clipped = (soc_raw != soc_new)

    # 4. 电网交换功率
    grid_power = p_load_eff - p_pv_eff + p_batt

    # 5. 变压器负载率
    q_load = p_load_eff * math.tan(math.acos(self.LOAD_PF))  # PF=0.90
    s_transformer = math.sqrt(grid_power**2 + (q_load - q_batt)**2)
    load_rate = s_transformer / self.TRANSFORMER_KVA

    # 6. 电压更新 (Grid2Op 或 VoltageSimulator)
    if self._use_grid2op:
        # Grid2Op 模式: 三相潮流计算
        va, vb, vc = self._grid2op_voltages
    else:
        # VoltageSimulator 降级模式
        p_net = p_pv_eff - p_load_eff + p_batt
        va, vb, vc = self.voltage_sim.step(p_net, q_batt, self.va, self.vb, self.vc)

    # 7. 需量更新 (15分钟滑动均值)
    self._update_demand(p_load_eff)

    return PhysicsResult(...)
```

### 3.5 Oracle 预测（LSTM 后备）

```python
class OraclePredictor:
    """使用真实未来数据 + 高斯噪声模拟预测, 噪声标准差 = 真实值 × 0.05。"""

    def predict_pv(self, step_idx: int) -> np.ndarray:
```

## 4. 环境仿真 — 电压仿真引擎（Grid2Op）

### 4.1 技术选型

**路线 A：组合模式（Composite Pattern）**

> `MupcEnv` 内部持有 `Grid2Op Env` 实例，Grid2Op 作为内部潮流计算引擎，不继承 Gymnasium 基类。

```
MupcEnv (外部: Gymnasium 接口)
  └── Grid2OpPowerFlow (内部: 潮流计算引擎, LightSimBackend)
        ├── PandapowerNetwork (农网台区物理拓扑)
        └── NumpyChronics (时序数据注入)
```

| 方案 | 描述 | 优点 | 缺点 | 选择 |
|------|------|------|------|------|
| 路线 A：组合模式 | MupcEnv 持有 Grid2Op 实例，外部保持 Gymnasium | 接口完全兼容，改动最小 | Grid2Op 需独立管理生命周期 | **采用** |
| 路线 B：继承模式 | MupcEnv 继承 Grid2Op 环境基类 | 直接利用 Grid2Op 生态 | Grid2Op 与 Gymnasium 基类冲突，适配成本高 | 放弃 |
| 路线 C：降级方案 | 保留原 VoltageSimulator，通过开关切换 | 改动最少 | 两种仿真引擎需同时维护，长期成本高 | 放弃 |

```python
def _expand_single_to_three_phase(self, scalar_value: float,
                                   phase_imbalance: float = 0.003
                                   ) -> np.ndarray:
    """将单相标量展开为三相数组，叠加轻微不平衡度。"""
    base = scalar_value / 3.0  # 平均分配到三相
    phase_a = base * (1.0 + np.random.uniform(-phase_imbalance, phase_imbalance))
    phase_b = base * (1.0 + np.random.uniform(-phase_imbalance, phase_imbalance))
    phase_c = base * (1.0 + np.random.uniform(-phase_imbalance, phase_imbalance))
    return np.array([phase_a, phase_b, phase_c])
```

```python
class MupcEnv(gym.Env):
    def __init__(self, data: dict, mode: str = "all",
                 lstm_predictor: Any = None,
                 reward_weights: dict[str, list[float]] | None = None,
                 use_grid2op: bool = True):  # ← 新增参数
        # ... 原有初始化逻辑 ...

        if use_grid2op:
            from grid2op_env import Grid2OpPowerFlow, NumpyChronics, create_mupc_network
            self._grid2op_power_flow: Grid2OpPowerFlow | None = None
            self._use_grid2op = True
        else:
            self._use_grid2op = False
            self._voltage_sim = VoltageSimulator()

    def _init_grid2op(self):
        """延迟初始化 Grid2Op（首次 reset 前不创建）。"""
        net = create_mupc_network()
        chronics = NumpyChronics(self._data)
        self._grid2op_power_flow = Grid2OpPowerFlow(
            net, chronics, storage_soc_init=self._soc
        )

    def reset(self, seed=None, options=None) -> tuple[np.ndarray, dict]:
        # ... 原有 reset 逻辑 ...

        if self._use_grid2op:
            if self._grid2op_power_flow is None:
                self._init_grid2op()
            self._grid2op_power_flow.reset(initial_storage_soc=self._soc)

        # ... 其余 reset 逻辑 ...

    def step(self, action: np.ndarray) -> tuple[np.ndarray, float, bool, bool, dict]:
        """执行一步仿真，三相电压由 Grid2Op/Pandapower 计算。"""

        # 1. Grid2Op SOC 同步（step 开始时：Grid2Op → MupcEnv）
        if self._use_grid2op:
            grid_soc = self._grid2op_power_flow.get_storage_soc()
            self._soc = grid_soc  # 同步 Grid2Op 的 SOC 到 MupcEnv

        # 2. Q_batt 实时电压环（与原逻辑一致）
        v_prev = (self._va + self._vb + self._vc) / 3.0
        q_batt = float(np.clip(-K_Q_V * (v_prev - 1.0),
                               -Q_BATT_MAX_KVAR, Q_BATT_MAX_KVAR))
        self._q_realtime_margin = 1.0 - (abs(q_batt) / Q_BATT_MAX_KVAR)

        # 3. 动作约束校验（ACT-01/03/05）
        clamped, violated, violations = self._validator.validate(
            action, dispatch_p_use, q_batt_real=q_batt)

        # 4. 反归一化动作
        p_batt = clamped[0] * P_BATT_MAX_KW
        load_shed = clamped[1] * LOAD_SHED_MAX_KW

        # 5. 有效负荷与光伏
        p_load_raw = float(self._data["load_power"][self._step_idx])
        p_load_eff = max(0.0, p_load_raw - load_shed)
        p_pv_raw = float(self._data["pv_power"][self._step_idx])
        p_pv_eff = p_pv_raw

        # 6. SOC 更新（MupcEnv 侧计算）
        soc_raw = self._soc + (-p_batt * DT_HOURS) / BATTERY_CAPACITY_KWH
        soc_new = float(np.clip(soc_raw, SOC_MIN, SOC_MAX))
        soc_clipped = abs(soc_raw - soc_new) > 1e-9

        # 7. Grid2Op 潮流计算（替换 VoltageSimulator）
        if self._use_grid2op:
            # 转换为 Grid2Op 单位: kW → MW
            storage_p_mw = p_batt / 1000.0
            storage_q_mvar = q_batt / 1000.0  # kVar → MVar

            va, vb, vc, has_illegal = self._grid2op_power_flow.step(
                storage_p_mw, storage_q_mvar)

            # 同步 SOC 到 Grid2Op（step 结束时：MupcEnv → Grid2Op）
            self._grid2op_power_flow.set_storage_soc(soc_new)
        else:
            # 降级到原 VoltageSimulator
            p_net = p_pv_eff - p_load_eff + p_batt
            va, vb, vc = self._voltage_sim.step(
                p_net, float(q_batt), self._va, self._vb, self._vc)
            has_illegal = False

        # 8. 后续逻辑与原实现一致（电网功率、变压器负载率、奖励计算等）
        ...
```

## 5. 环境仿真 — 奖励函数

### 5.1 设计概述

奖励函数严格对齐 MUPC AI 引擎 PRD 第 6 章。每种场景的公式独立实现，通过 `current_mode` 选择。返回值已归一化处理供 RL 训练使用。

### 5.2 SCENE-01：台区季节性负荷模式

> **v2.6 变更：** 动作空间恢复为 3 维（新增 pv_limit 主动弃光）；SCENE-01 新增 w6 电压斜率惩罚 R_slope = w6·|ΔV|；动作值域全面修正（P_batt [-50,50]，load_shedding [0,60]，ACT-03 S_max 200kVA）。

```python
def _reward_agri(self, r: PhysicsResult, prev_p_batt: float) -> dict:
    """R_agri = w1*R_pv_consumption - w2*P_battery_degradation - w3*P_transformer_overload
                - w4*P_voltage_deviation - w5*R_ramp - w6*R_slope"""

    # R_pv_consumption: 光伏消纳率 [0, 1]
    pv_total = self.p_pv_raw
    pv_self = min(pv_total, self.p_load_raw) + max(0, -r.p_batt)  # 负荷消纳 + 充电消纳
    r_pv = min(pv_self / max(pv_total, 1e-6), 1.0)

    # P_battery_degradation: 电池衰减 [0, 1]
    delta_soc = abs(self.soc - r.soc_new)
    p_batt_deg = delta_soc  # 简化: alpha=1.0

    # P_transformer_overload: 过载惩罚 [0, ∞)
    p_overload = max(0, r.load_rate - 1.0)

    # P_voltage_deviation: 电压质量惩罚（死区 ±5%，越限连续 2 步触发）
    v_avg = (self.va + self.vb + self.vc) / 3.0
    p_voltage = self._voltage_penalty_with_deadband(v_avg)

    # R_ramp: 功率变化率惩罚（归一化到 C-rate 变化率）
    delta_p = abs(r.p_batt - prev_p_batt)
    r_ramp = delta_p / self.BATTERY_CAPACITY_KWH  # kW/kWh = 1/h

    # R_slope: 电压变化斜率惩罚（v2.6 新增，迫使 AI 平滑调节）
    delta_v = abs(r.v_avg - self._prev_v_avg)
    r_slope = delta_v

    # 默认权重: w1=1.0, w2=0.5, w3=2.0, w4=1.0, w5=0.5, w6=0.0
    total = (self.w[0] * r_pv
             - self.w[1] * p_batt_deg
             - self.w[2] * p_overload
             - self.w[3] * p_voltage
             - self.w[4] * r_ramp
             - self.w[5] * r_slope)

    return {
        "total": total,
        "r_pv_consumption": r_pv,
        "p_battery_degradation": -p_batt_deg,
        "p_transformer_overload": -p_overload,
        "p_voltage_deviation": -p_voltage,
        "r_ramp": -r_ramp,
        "r_slope": -r_slope,  # v2.6 新增
    }

def _voltage_penalty_with_deadband(self, v_avg: float) -> float:
    """电压质量惩罚：死区 ±5%，越限连续 2 步才触发"""
    V_DEAD = 0.05   # ±5% 死区
    V_REF = 1.0
    dev = abs(v_avg - V_REF)
    if dev <= V_DEAD:
        return 0.0  # 死区内无惩罚
    dev_excess = dev - V_DEAD
    if v_avg < V_REF - V_DEAD:
        # 低电压，斜率更高（2.0）
        return 2.0 * (dev_excess / 0.10) ** 2
    else:
        # 高电压（1.0）
        return 1.0 * (dev_excess / 0.10) ** 2
```

### 5.3 SCENE-B1：自主套利

```python
def _reward_arbitrage(self, r: PhysicsResult) -> dict:
    """R_arbitrage = w1 * R_price_spread - w2 * P_battery_degradation"""

    # 价差套利: p_batt > 0(放电) × 高电价 → 正收益
    price = self.data['current_price'][self.step_idx]
    avg_price = 0.8  # 参考平均电价
    r_spread = r.p_batt * (price - avg_price) / (self.P_BATT_MAX_KW * 0.4)  # [-1,1]（v2.6: P_BATT_MAX_KW=50）

    # 电池衰减 (同上)
    p_batt_deg = abs(self.soc - r.soc_new)

    total = self.w[0] * r_spread - self.w[1] * p_batt_deg
    return {"total": total, "r_price_spread": r_spread, "p_battery_degradation": -p_batt_deg}
```

```python
def _reward_demand(self, r: PhysicsResult) -> dict:
    """R_demand = w1 * R_demand_penalty_avoidance - w2 * P_comfort_loss"""

    # 需量减免: 峰值低于基线即奖励
    d_peak_baseline = self.contract_demand  # 300 kW
    d_peak_actual = self.current_demand
```

### 5.5 SCENE-B3：虚拟电厂

```python
def _reward_vpp(self, r: PhysicsResult) -> dict:
    """R_vpp = w1 * R_ancillary_service + w2 * R_response_accuracy - w3 * P_deadline_deviation"""

    dispatch_p = self.data['dispatch_p'][self.step_idx] or 0.0

    # 辅助服务收益 (容量+里程简化)
    capacity_price = 0.1  # 元/kW
    mileage_price = 0.05   # 元/kW
    r_ancillary = (abs(dispatch_p) * capacity_price
                   + abs(r.p_batt - self.battery_power_prev) * mileage_price)
    r_ancillary = np.clip(r_ancillary / 50.0, 0, 1)  # 归一化

    # 响应精度 (与调度指令的偏差)
    if abs(dispatch_p) > 1e-6:
        r_accuracy = max(0, 1 - abs(r.p_batt - dispatch_p) / abs(dispatch_p))
    else:
        r_accuracy = 1.0  # 无指令时满分

    # 响应延迟简化: 使用功率变化率作为代理
    p_deadline = abs(r.p_batt - self.battery_power_prev) / (50 * self.DT_HOURS)
    p_deadline = min(p_deadline, 1.0)

    total = self.w[0] * r_ancillary + self.w[1] * r_accuracy - self.w[2] * p_deadline
    return {"total": total, "r_ancillary_service": r_ancillary,
            "r_response_accuracy": r_accuracy, "p_deadline_deviation": -p_deadline}
```

```python
def _reward_green(self, r: PhysicsResult) -> dict:
    """R_green = w1 * R_green_consumption + w2 * R_carbon_reduction"""

    # 绿电消纳比例
    p_pv = self.data['pv_power'][self.step_idx]
    p_load_eff = self.p_load_raw - r.load_shedding
    e_total = max(p_load_eff, 1e-6)
    e_green = min(p_pv, p_load_eff) + max(0, -r.p_batt)  # 负荷消纳 + 充电
    r_green_consumption = min(e_green / e_total, 1.0)

    # 碳排放减少
    grid_power_buy = max(0, r.grid_power)  # 仅买电产生碳排放
    c_baseline = p_load_eff * self.GRID_EMISSION_FACTOR * self.DT_HOURS  # 全从电网
    c_actual = grid_power_buy * self.GRID_EMISSION_FACTOR * self.DT_HOURS
    r_carbon = max(0, (c_baseline - c_actual) / max(c_baseline, 1e-6))

    total = self.w[0] * r_green_consumption + self.w[1] * r_carbon
    return {"total": total, "r_green_consumption": r_green_consumption, "r_carbon_reduction": r_carbon}
```

## 6. LSTM 预测模型

### 6.1 模型架构

```python
class LSTMForecast(nn.Module):
    """2 层 LSTM + Linear head: 光伏+负荷联合预测。

    Input:  (batch, seq_len=4, features=6)
            features = [pv, load, ghi, temp, sin_h, cos_h]
    Output: (batch, 30) = [pv_1..pv_15, load_1..load_15]
    """

    def __init__(self, input_dim=6, hidden_dim=64, num_layers=2,
                 forecast_steps=15, dropout=0.1):
        super().__init__()
        self.lstm = nn.LSTM(input_dim, hidden_dim, num_layers,
                            batch_first=True, dropout=dropout)
        self.head_pv = nn.Linear(hidden_dim, forecast_steps)    # 15 维光伏
        self.head_load = nn.Linear(hidden_dim, forecast_steps)  # 15 维负荷

    def forward(self, x):
        """
        x: (batch, 4, 6)
        Returns: (batch, 30) — [pv_forecast(15), load_forecast(15)]
        """
        _, (h_n, _) = self.lstm(x)           # h_n: (2, batch, 64)
        last_hidden = h_n[-1]                 # (batch, 64)
        pv_pred = self.head_pv(last_hidden)   # (batch, 15)
        load_pred = self.head_load(last_hidden)
        return torch.cat([pv_pred, load_pred], dim=-1)
```

### 6.3 Oracle 后备

```python
class OraclePredictor:
    """LSTM 模型不可用时提供预测: 真实未来值 + 噪声。
    噪声幅度: 光伏 5%, 负荷 8%, 模拟预测误差。
    输出维度与 LSTM 一致 (15 维 × 2)。
    """
    PV_NOISE = 0.05
    LOAD_NOISE = 0.08

    def predict(self, step_idx: int) -> np.ndarray:
        """返回 (30,) = [pv_15, load_15]"""
        pv_future = self.pv_data[step_idx+1 : step_idx+16]
        load_future = self.load_data[step_idx+1 : step_idx+16]
        pv_pred = pv_future * (1 + np.random.normal(0, self.PV_NOISE, 15))
        load_pred = load_future * (1 + np.random.normal(0, self.LOAD_NOISE, 15))
        return np.concatenate([np.clip(pv_pred, 0, None), np.clip(load_pred, 0, None)])
```

## 7. 动作约束校验

### 7.1 类设计

> **v2.6 变更：** ACT-04（pv_limit ∈ [0,1]）约束规则恢复；动作空间恢复为 3 维（p_batt, load_shedding, pv_limit）；P_BATT_MAX 500→50kW，S_MAX 500→200kVA，LOAD_SHED_MAX 500→60kW。

```python
class ActionValidator:
    """4 条动作约束规则, 顺序执行 clamp。"""

    P_BATT_MAX = 50.0          # kW（匹配电池最大充放电功率）
    S_MAX = 200.0              # kVA (功率圆上限，匹配变压器 200kVA)
    LOAD_SHED_MAX = 60.0       # kW（匹配负荷峰值）
    DELTA_P_MAX = 50.0         # kW/步（电池变化率保护）

    def __init__(self, p_batt_max: float = 50.0, s_max: float = 200.0,
                 load_shed_max: float = 60.0, delta_p_max: float = 50.0):
        self.P_BATT_MAX = p_batt_max
        self.S_MAX = s_max
        self.LOAD_SHED_MAX = load_shed_max
        self.DELTA_P_MAX = delta_p_max
        self.prev_p_batt = 0.0

    def validate(self, action: np.ndarray, dispatch_p: float | None,
                 q_batt_real: float = 0.0
                 ) -> tuple[np.ndarray, bool, dict]:
        """
        Args:
            action: [p_batt_norm, load_shed_norm, pv_limit_norm]  # 3 维
            dispatch_p: 调度指令 或 None
            q_batt_real: 实时控制核心闭环的 Q 值（用于功率圆校验）
        Returns:
            (clamped_action, violated, violations)
        """
        violations = {}
        p, ls, pv_limit = self._denormalize(action)

        # ACT-01: Δp_batt ≤ 50 kW
        if abs(p - self.prev_p_batt) > self.DELTA_P_MAX:
            p = self.prev_p_batt + np.clip(
                p - self.prev_p_batt, -self.DELTA_P_MAX, self.DELTA_P_MAX)
            violations['ACT-01'] = True

        # ACT-03: sqrt(p² + q_real²) ≤ 200 kVA（Q 由实时控制闭环，v2.6 S_MAX=200）
        s = math.sqrt(p**2 + q_batt_real**2)
        if s > self.S_MAX:
            scale = self.S_MAX / s
            p *= scale
            violations['ACT-03'] = True

        # ACT-04: pv_limit ∈ [0, 1]（v2.6 新增）
        pv_limit = float(np.clip(pv_limit, 0.0, 1.0))
        if pv_limit != action[2]:
            violations['ACT-04'] = True

        # ACT-05: dispatch_p 有效时 |p_batt| ≤ |dispatch_p|
        if dispatch_p is not None and abs(dispatch_p) > 1e-6:
            if abs(p) > abs(dispatch_p):
                p = np.clip(p, -abs(dispatch_p), abs(dispatch_p))
                violations['ACT-05'] = True

        # 更新历史
        self.prev_p_batt = p

        return self._renormalize(p, ls, pv_limit), bool(violations), violations

    def reset(self):
        self.prev_p_batt = 0.0
```

### 7.2.1 下垂模式（v2.7 新增）

当 `config.dual_control.enabled=true` 时，RL 策略输出 4 维动作，执行器根据下垂公式计算最终功率：

```
P_output = P_ref + k_droop × ΔV
```

| 维度 | 字段 | 激活函数 | 反归一化 | 说明 |
|------|------|----------|----------|------|
| A1 | P_ref | Tanh | [-1, 1] → [-50, 50] kW | 有功功率基准点 |
| A2 | k_droop | Tanh | [-1, 1] → [k_min, k_max] kW/V | 电压-有功下垂系数 |
| A3 | load_shedding | Sigmoid | [0, 1] → [0, 60] kW | 可中断负荷切除量 |
| A4 | pv_limit | Sigmoid | [0, 1] → [0.1, 1] | 光伏限功率比例（防逆流） |

**配置参数：**

| 参数 | 默认值 | 说明 |
|------|--------|------|
| `dual_control.enabled` | `false` | 启用/禁用下垂模式（默认关闭，维持 3 维输出） |
| `dual_control.k_droop_min` | `-100.0` | 下垂系数下界 (kW/V) |
| `dual_control.k_droop_max` | `100.0` | 下垂系数上界 (kW/V) |
| `dual_control.p_ref_ramp_limit_kw` | `50.0` | P_ref 单步变化率限制 (kW/步) |
| `dual_control.pv_limit_min` | `0.1` | 光伏限功率下限（防逆流） |

**ACT-DUAL 约束规则：**

| 规则 | 约束条件 | 越限处理 |
|------|----------|----------|
| ACT-DUAL-01 | P_ref ∈ [-50, 50] kW | clamp |
| ACT-DUAL-02 | k_droop ∈ [k_droop_min, k_droop_max] | clamp |
| ACT-DUAL-03 | ΔP_ref ≤ p_ref_ramp_limit_kw (50 kW/步) | 限制变化率 |
| ACT-DUAL-04 | \|P_ref\| ≤ \|dispatch_p\| | clamp |
| ACT-DUAL-05 | pv_limit ≥ 0.1（防逆流） | clamp |

**执行器计算公式：**

```python
def compute_p_output(p_ref: float, k_droop: float, delta_v: float,
                     prev_p_output: float = 0.0) -> float:
    """计算最终电池功率输出。

    Args:
        p_ref: 有功功率基准点 (kW)
        k_droop: 下垂系数 (kW/V)
        delta_v: 电压偏差 = V_real - V_nominal (标幺值)
        prev_p_output: 上一步实际输出功率（用于 ramp 限制）

    Returns:
        最终电池功率设定值 (kW)
    """
    # 1. 下垂控制
    p_droop = k_droop * delta_v

    # 2. 基准 + 下垂
    p_output = p_ref + p_droop

    # 3. Ramp 限制
    delta_p = p_output - prev_p_output
    if abs(delta_p) > 50.0:
        p_output = prev_p_output + np.sign(delta_p) * 50.0

    return p_output
```

```python
        mean_actions = torch.cat([
```

```python
def compute_gae(rewards, values, dones, gamma=0.99, lam=0.95):
    advantages = np.zeros_like(rewards, dtype=np.float32)
    last_adv = 0.0
    for t in reversed(range(len(rewards))):
        if dones[t]:
            delta = rewards[t] - values[t]
            last_adv = 0.0
        else:
            delta = rewards[t] + gamma * values[t+1] - values[t]
        advantages[t] = last_adv = delta + gamma * lam * last_adv
    returns = advantages + values[:-1]
    return advantages, returns
```

## 14. 技术决策记录

### TD-01：训练环境合成缺失字段

- **决策**：对于部署端由实时数据源提供的字段（电压、需量、调度指令），训练环境使用仿真模型合成
- **理由**：SMART-DS 数据集仅含光伏/负荷/气象，其余 17 个字段无法从历史数据获取
- **影响**：合成数据的分布可能与真实 MUPC 硬件存在差异。部署前需使用真实数据微调

### TD-02：58 维 vs 59 维输入

- **决策**：单模式训练使用 58 维（匹配部署端 `to_input_vector()`）；多模式训练追加 mode_id 为第 59 维
- **理由**：多模式需要策略感知当前场景。部署端 ModeSelector 在运行时提供场景 ID，可追加到向量
- **影响**：多模式 ONNX 模型输入为 59 维，部署端 `to_input_vector()` 已在 v2.5 扩展至 56 维，多模式追加 mode_id 为 57 维，与训练端 58 维一致

### TD-03：LSTM 训练在管线内

- **决策**：LSTM 预测模型在本训练管线中训练，而非作为外部依赖
- **理由**：SMART-DS 数据可用于训练 LSTM，管线需要预测值作为 RL 状态输入
- **影响**：增加了 PyTorch 依赖。但 ONNX 导出本身就需要 PyTorch

### TD-04：训练步长保持 15 分钟

- **决策**：训练时间步长保持 SMART-DS数据的 15 分钟分辨率，不升级到 1 秒
- **理由**：RL 训练在 15 分钟粒度下已可学到有效策略。部署端 1 秒周期复用同一策略输出
- **影响**：动作约束规则 ACT-01/02需按 `Δ/步` 而非 `Δ/秒` 理解

### TD-05：电压仿真技术选型（v2.3 新增）

- **决策**：采用 Grid2Op + Pandapower 三相潮流作为主要电压仿真引擎，VoltageSimulator 作为降级方案
- **理由**：真实三相潮流计算精度远高于简化 Q-V 灵敏度系数，三相不平衡度自然呈现
- **影响**：增加了 grid2op、pandapower 依赖；每步仿真耗时可能增加（通过 lightsim2grid 缓解）

### TD-06：分层控制架构（v2.4）

- **决策**：Q 控制（q_batt_set）交由实时控制核心根据电压闭环调节，RL 仅输出 P_batt + Load_shedding
- **理由**：300kW 错峰启停产生高频随机脉冲，Q 是 ms 级快变量与 P 是 min 级慢变量，单一网络同时学习两个时间尺度任务收敛困难且易振荡
- **影响**：RL 专注能量管理（光伏消纳、SOC 平衡、过载预防），电压质量由底层保障。训练环境动作空间从 4 维缩减为 2 维

### TD-07：电压死区 + 功率变化率惩罚

- **决策**：MODE-01 奖励函数新增 ±5% 电压死区（越限连续 2 步触发）和 R_ramp 功率变化率惩罚（归一化到 C-rate）
- **理由**：防止 AI 对瞬时高频脉冲过度响应；变化率惩罚延长电池寿命，避免大功率场景惩罚量级失衡
- **影响**：MODE-01 权重表新增 w4（电压质量）、w5（功率变化率）

### TD-08：Grid2Op 组合模式（v2.3 新增）

- **决策**：MupcEnv 内部持有 Grid2Op 实例（组合模式），而非继承 Grid2Op 环境基类（继承模式）
- **理由**：Grid2Op 与 Gymnasium 基类不兼容，无法多重继承；组合模式使 Grid2Op 完全内部化，不影响 RL 训练接口
- **影响**：Grid2Op 生命周期（初始化/reset/step/close）需在 MupcEnv 内部显式管理

### TD-09：SOC 双向同步（v2.3 新增）

- **决策**：MupcEnv._soc 与 Grid2Op storage.soc 每步双向同步
- **理由**：Grid2Op 管理自己的储能 SOC 状态，MupcEnv 也需要独立 SOC 用于观测构建和奖励计算，两者必须一致
- **影响**：每步 step() 增加两次 SOC 同步操作；潮流不收敛时 SOC 同步照常进行

### TD-10：潮流不收敛回退策略（v2.3 新增）

- **决策**：潮流计算不收敛或电压为 NaN 时，回退到上一时刻安全电压
- **理由**：潮流不收敛不等于 episode 结束，应保持安全值继续训练
- **影响**：`has_illegal=True` 标记会被设置，但不触发 terminated/truncated
